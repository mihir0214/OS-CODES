#include <stdio.h>

int main()
{
    int n, i;
    float s = 0, t = 0;

    int pid[20], at[20], bt[20];
    int ct[20], wt[20], tat[20], rt[20];
    int completed = 0;
    int time = 0;
    int shortest, min;
    float avg_wt = 0, avg_tat = 0;

    printf("Enter the number of processes: ");
    scanf("%d", &n);

    printf("Enter Arrival Time for each process:\n");

    for (i = 0; i < n; i++) {
        pid[i] = i + 1;

        printf("P%d: ", i + 1);
        scanf("%d", &at[i]);
    }

    printf("Enter Burst Time for each process:\n");

    for (i = 0; i < n; i++) {
        pid[i] = i + 1;

        printf("P%d: ", i + 1);
        scanf("%d", &bt[i]);
    }

   
    for (i = 0; i < n; i++)
    {
        rt[i] = bt[i];
    }

    // SRTF Logic
    while (completed < n)
    {
        shortest = -1;
        min = 9999;

        // Find process with shortest remaining time
        for (i = 0; i < n; i++)
        {
            if (at[i] <= time && rt[i] > 0)
            {
                if (rt[i] < min)
                {
                    min = rt[i];
                    shortest = i;
                }
            }
        }

        // If no process has arrived
        if (shortest == -1)
        {
            time++;
        }
        else
        {
            // Execute process for 1 unit
            rt[shortest]--;
            time++;

            // Process completed
            if (rt[shortest] == 0)
            {
                completed++;

                ct[shortest] = time;
               
                tat[shortest] = ct[shortest] - at[shortest];

                wt[shortest] = tat[shortest] - bt[shortest];

                s += wt[shortest];
                t += tat[shortest];
            }
        }
    }

    avg_wt = s / (float)n;
    avg_tat = t / (float)n;

    printf("\nProcess\tAT\tBT\tCT\tTAT\tWT\n");

    for (i = 0; i < n; i++)
    {
        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               pid[i],
               at[i],
               bt[i],
               ct[i],
               tat[i],
               wt[i]);
    }

    printf("\nAverage Waiting Time = %.2f\n", avg_wt);
    printf("Average Turnaround Time = %.2f\n", avg_tat);

    return 0;
}
