#!/bin/bash

echo "Enter first number:"
read a

echo "Enter second number:"
read b

echo "----- MENU -----"
echo "1. Addition"
echo "2. Subtraction"
echo "3. Multiplication"
echo "4. Division"
echo "5. Exit"

echo "Enter your choice:"
read ch

case $ch in
    1)
        echo "Addition = $((a + b))"
        ;;
    2)
        echo "Subtraction = $((a - b))"
        ;;
    3)
        echo "Multiplication = $((a * b))"
        ;;
    4)
        if [ $b -eq 0 ]
        then
            echo "Division by zero is not possible"
        else
            echo "Division = $((a / b))"
        fi
        ;;
    5)
        echo "Exiting..."
        ;;
    *)
        echo "Invalid choice"
        ;;
esac
